export function Radio({checked,onChange,label,description,style}){
  return <label onClick={()=>onChange&&onChange()} style={{display:'flex',alignItems:'flex-start',gap:10,cursor:'pointer',fontFamily:'var(--font-body)',...style}}>
    <span style={{width:22,height:22,borderRadius:'50%',flexShrink:0,border:checked?'7px solid var(--action-primary)':'2px solid var(--border-strong)',background:'var(--surface-card)',boxSizing:'border-box',transition:'border var(--dur-fast) var(--ease-out)'}}></span>
    <span><span style={{fontWeight:700,fontSize:14,display:'block'}}>{label}</span>
    {description&&<span style={{fontWeight:600,fontSize:12,color:'var(--text-muted)'}}>{description}</span>}</span>
  </label>;
}
