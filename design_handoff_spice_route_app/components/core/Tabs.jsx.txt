export function Tabs({tabs=[],active,onChange,style}){
  return <div style={{display:'flex',gap:4,background:'var(--cream-200)',borderRadius:'var(--radius-pill)',padding:4,...style}}>
    {tabs.map(t=><button key={t} onClick={()=>onChange&&onChange(t)}
      style={{flex:1,border:'none',borderRadius:'var(--radius-pill)',padding:'9px 16px',font:'700 13px var(--font-body)',cursor:'pointer',background:active===t?'var(--surface-card)':'transparent',color:active===t?'var(--text-brand)':'var(--text-muted)',boxShadow:active===t?'var(--shadow-card)':'none',transition:'all var(--dur-base) var(--ease-out)'}}>{t}</button>)}
  </div>;
}
