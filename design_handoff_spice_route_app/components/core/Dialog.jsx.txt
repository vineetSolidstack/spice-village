export function Dialog({open,onClose,title,children,footer,style}){
  if(!open)return null;
  return <div onClick={onClose} style={{position:'fixed',inset:0,background:'rgba(43,29,18,.45)',display:'flex',alignItems:'flex-end',justifyContent:'center',zIndex:100}}>
    <div onClick={e=>e.stopPropagation()} style={{background:'var(--surface-card)',borderRadius:'var(--radius-xl) var(--radius-xl) 0 0',padding:20,width:'100%',maxWidth:430,boxShadow:'var(--shadow-overlay)',fontFamily:'var(--font-body)',...style}}>
      <div style={{width:40,height:4,borderRadius:2,background:'var(--cream-300)',margin:'0 auto 14px'}}></div>
      {title&&<h3 style={{fontFamily:'var(--font-display)',fontWeight:700,fontSize:18,margin:'0 0 10px'}}>{title}</h3>}
      <div>{children}</div>
      {footer&&<div style={{marginTop:16,display:'flex',gap:10}}>{footer}</div>}
    </div>
  </div>;
}
