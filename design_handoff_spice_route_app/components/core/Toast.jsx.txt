export function Toast({tone='success',children,style}){
  const t={success:{bg:'var(--status-success)',},danger:{bg:'var(--status-danger)'},info:{bg:'var(--cocoa-900)'}}[tone];
  return <div style={{background:t.bg,color:'#fff',borderRadius:'var(--radius-md)',padding:'12px 16px',font:'700 14px var(--font-body)',boxShadow:'var(--shadow-raised)',display:'inline-flex',alignItems:'center',gap:8,...style}}>{children}</div>;
}
